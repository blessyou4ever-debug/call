cmake_minimum_required(VERSION 3.22.1)
project(callwhisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
FetchContent_Declare(whisper URL https://codeload.github.com/ggml-org/whisper.cpp/tar.gz/refs/tags/v1.7.6 URL_HASH SHA256=166140e9a6d8a36f787a2bd77f8f44dd64874f12dd8359ff7c1f4f9acb86202e)
FetchContent_MakeAvailable(whisper)
add_library(callwhisper SHARED whisper_jni.cpp)
target_link_libraries(callwhisper PRIVATE whisper android log)
target_link_options(callwhisper PRIVATE "-Wl,-z,max-page-size=16384")
